cmake_minimum_required(VERSION 3.16)
project(keyboard_synth VERSION 0.1.0 LANGUAGES C)

option(CSYNTH_BUILD_APP "Build the SDL synth window" ON)
option(CSYNTH_BUILD_EXAMPLES "Build the small libcsynth examples" OFF)
option(CSYNTH_BUILD_SDL_ADAPTER "Build the optional SDL audio adapter" ${CSYNTH_BUILD_APP})
if(CSYNTH_BUILD_APP AND NOT CSYNTH_BUILD_SDL_ADAPTER)
    message(FATAL_ERROR "The SDL app needs CSYNTH_BUILD_SDL_ADAPTER=ON")
endif()
include(CTest)
include(GNUInstallDirs)
include(CMakePackageConfigHelpers)

function(csynth_compile_options target)
    set_target_properties(${target} PROPERTIES C_EXTENSIONS OFF)
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4)
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
    endif()
endfunction()

# The reusable audio engine. No window, font, file browser, or MIDI player here.
add_library(csynth STATIC synth.c effects.c envelope.c synth_config.c oscillator.c fm.c presets.c)
add_library(csynth::csynth ALIAS csynth)
target_compile_features(csynth PUBLIC c_std_11)
target_include_directories(csynth PUBLIC
    $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}>
    $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}/csynth>)
if(UNIX)
    target_link_libraries(csynth PUBLIC m)
endif()
set_target_properties(csynth PROPERTIES POSITION_INDEPENDENT_CODE ON)
csynth_compile_options(csynth)

if(CSYNTH_BUILD_SDL_ADAPTER)
    find_package(SDL2 2.0.18 REQUIRED CONFIG)
    add_library(csynth_sdl STATIC synth_sdl.c)
    add_library(csynth::sdl ALIAS csynth_sdl)
    target_link_libraries(csynth_sdl PUBLIC csynth::csynth SDL2::SDL2)
    csynth_compile_options(csynth_sdl)
endif()

if(CSYNTH_BUILD_APP)
    if(APPLE)
        set(CMAKE_FIND_FRAMEWORK LAST)
    endif()
    find_package(Freetype REQUIRED)
    add_executable(keyboard_synth main.c options.c spectrum.c midi.c file_browser.c spectrogram.c)
    target_link_libraries(keyboard_synth PRIVATE csynth::sdl Freetype::Freetype)
    if(TARGET SDL2::SDL2main)
        target_link_libraries(keyboard_synth PRIVATE SDL2::SDL2main)
    endif()
    target_compile_definitions(keyboard_synth PRIVATE SYNTH_DEFAULT_PRESET_DIR="${CMAKE_CURRENT_SOURCE_DIR}/presets")
    csynth_compile_options(keyboard_synth)
endif()

if(CSYNTH_BUILD_EXAMPLES)
    add_executable(csynth_live examples/live.c)
    target_link_libraries(csynth_live PRIVATE csynth::csynth)
    csynth_compile_options(csynth_live)
endif()

install(TARGETS csynth EXPORT csynthTargets ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
install(FILES synth.h synth_config.h fm.h oscillator.h envelope.h effects.h presets.h
        DESTINATION ${CMAKE_INSTALL_INCLUDEDIR}/csynth)
install(EXPORT csynthTargets NAMESPACE csynth:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/csynth)
configure_package_config_file(cmake/csynthConfig.cmake.in
    ${CMAKE_CURRENT_BINARY_DIR}/csynthConfig.cmake
    INSTALL_DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/csynth)
write_basic_package_version_file(${CMAKE_CURRENT_BINARY_DIR}/csynthConfigVersion.cmake
    VERSION ${PROJECT_VERSION} COMPATIBILITY SameMajorVersion)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/csynthConfig.cmake
              ${CMAKE_CURRENT_BINARY_DIR}/csynthConfigVersion.cmake
        DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/csynth)
install(FILES docs/libcsynth-api.md DESTINATION ${CMAKE_INSTALL_DOCDIR})
install(DIRECTORY presets/factory DESTINATION ${CMAKE_INSTALL_DATADIR}/csynth/presets)

if(BUILD_TESTING)
    add_executable(test_factory_bank tests/test_factory_bank.c)
    target_compile_definitions(test_factory_bank PRIVATE FACTORY_TEST_DIR="${CMAKE_CURRENT_SOURCE_DIR}/presets")
    add_executable(test_presets tests/test_presets.c)
    add_executable(test_midi tests/test_midi.c midi.c file_browser.c)
    add_executable(test_effects tests/test_effects.c)
    add_executable(test_envelope tests/test_envelope.c)
    add_executable(test_fm tests/test_fm.c)
    add_executable(test_options tests/test_options.c options.c)
    add_executable(test_spectrum tests/test_spectrum.c spectrum.c)
    add_executable(test_engine tests/test_engine.c)
    foreach(test_target IN ITEMS test_fm test_options test_engine test_envelope test_effects test_presets test_factory_bank)
        target_link_libraries(${test_target} PRIVATE csynth::csynth)
    endforeach()
    foreach(test_target IN ITEMS test_midi test_spectrum)
        target_include_directories(${test_target} PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
        target_compile_features(${test_target} PRIVATE c_std_11)
        if(UNIX)
            target_link_libraries(${test_target} PRIVATE m)
        endif()
    endforeach()
    foreach(test_name IN ITEMS presets factory_bank midi effects envelope fm options spectrum engine)
        csynth_compile_options(test_${test_name})
        add_test(NAME ${test_name} COMMAND test_${test_name})
    endforeach()
    if(CSYNTH_BUILD_SDL_ADAPTER)
        add_executable(test_synth tests/test_synth.c spectrum.c)
        target_link_libraries(test_synth PRIVATE csynth::sdl)
        target_compile_definitions(test_synth PRIVATE SDL_MAIN_HANDLED)
        csynth_compile_options(test_synth)
        add_test(NAME synth COMMAND test_synth)
    endif()
    if(CSYNTH_BUILD_APP)
        add_executable(test_app tests/test_app.c options.c spectrum.c midi.c file_browser.c spectrogram.c)
        # test_gui includes spectrogram.c to inspect event dispatch state.
        add_executable(test_gui tests/test_gui.c midi.c file_browser.c spectrum.c)
        target_compile_definitions(test_gui PRIVATE SDL_MAIN_HANDLED)
        foreach(test_target IN ITEMS test_gui test_app)
            target_link_libraries(${test_target} PRIVATE csynth::sdl Freetype::Freetype)
            csynth_compile_options(${test_target})
        endforeach()
        add_test(NAME gui COMMAND test_gui)
        add_test(NAME app COMMAND test_app)
        set_tests_properties(app PROPERTIES TIMEOUT 10)
        add_test(NAME help COMMAND keyboard_synth --help)
    endif()
endif()
