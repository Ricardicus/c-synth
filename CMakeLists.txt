cmake_minimum_required(VERSION 3.16)
project(keyboard_synth VERSION 0.1.0 LANGUAGES C)

find_package(SDL2 2.0.18 REQUIRED CONFIG)
if(APPLE)
    set(CMAKE_FIND_FRAMEWORK LAST)
endif()
find_package(Freetype REQUIRED)

add_executable(keyboard_synth main.c synth.c effects.c envelope.c synth_config.c oscillator.c fm.c options.c spectrum.c presets.c midi.c file_browser.c spectrogram.c)
target_compile_features(keyboard_synth PRIVATE c_std_11)
set_target_properties(keyboard_synth PROPERTIES C_EXTENSIONS OFF)
if(TARGET SDL2::SDL2main)
    target_link_libraries(keyboard_synth PRIVATE SDL2::SDL2main)
endif()

target_link_libraries(keyboard_synth PRIVATE SDL2::SDL2 Freetype::Freetype)
target_compile_definitions(keyboard_synth PRIVATE SYNTH_DEFAULT_PRESET_DIR="${CMAKE_CURRENT_SOURCE_DIR}/presets")
if(UNIX)
    target_link_libraries(keyboard_synth PRIVATE m)
endif()
if(MSVC)
    target_compile_options(keyboard_synth PRIVATE /W4)
else()
    target_compile_options(keyboard_synth PRIVATE -Wall -Wextra -Wpedantic)
endif()

include(CTest)
if(BUILD_TESTING)
    add_executable(test_factory_bank tests/test_factory_bank.c presets.c synth_config.c fm.c oscillator.c envelope.c effects.c)
    target_compile_definitions(test_factory_bank PRIVATE FACTORY_TEST_DIR="${CMAKE_CURRENT_SOURCE_DIR}/presets")
    add_executable(test_presets tests/test_presets.c presets.c synth_config.c fm.c oscillator.c)
    add_executable(test_midi tests/test_midi.c midi.c file_browser.c)
    add_executable(test_effects tests/test_effects.c effects.c synth_config.c fm.c oscillator.c)
    add_executable(test_envelope tests/test_envelope.c envelope.c)
    add_executable(test_fm tests/test_fm.c fm.c oscillator.c)
    add_executable(test_options tests/test_options.c options.c synth_config.c fm.c oscillator.c)
    add_executable(test_spectrum tests/test_spectrum.c spectrum.c)
    add_executable(test_synth tests/test_synth.c synth.c effects.c envelope.c synth_config.c fm.c oscillator.c spectrum.c)
    add_executable(test_app tests/test_app.c synth.c effects.c envelope.c synth_config.c fm.c oscillator.c
                   options.c spectrum.c presets.c midi.c file_browser.c spectrogram.c)
    add_executable(test_gui tests/test_gui.c presets.c midi.c file_browser.c synth.c effects.c envelope.c synth_config.c fm.c oscillator.c spectrum.c)
    target_link_libraries(test_gui PRIVATE SDL2::SDL2 Freetype::Freetype)
    target_compile_definitions(test_gui PRIVATE SDL_MAIN_HANDLED)
    target_link_libraries(test_synth PRIVATE SDL2::SDL2)
    target_link_libraries(test_app PRIVATE SDL2::SDL2 Freetype::Freetype)
    target_compile_definitions(test_synth PRIVATE SDL_MAIN_HANDLED)
    foreach(test_target IN ITEMS test_fm test_options test_spectrum test_synth test_app test_gui test_envelope test_effects test_presets test_midi test_factory_bank)
        target_include_directories(${test_target} PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
        target_compile_features(${test_target} PRIVATE c_std_11)
        set_target_properties(${test_target} PROPERTIES C_EXTENSIONS OFF)
        if(UNIX)
            target_link_libraries(${test_target} PRIVATE m)
        endif()
        if(MSVC)
            target_compile_options(${test_target} PRIVATE /W4)
        else()
            target_compile_options(${test_target} PRIVATE -Wall -Wextra -Wpedantic)
        endif()
    endforeach()
    add_test(NAME presets COMMAND test_presets)
    add_test(NAME factory_bank COMMAND test_factory_bank)
    add_test(NAME midi COMMAND test_midi)
    add_test(NAME effects COMMAND test_effects)
    add_test(NAME envelope COMMAND test_envelope)
    add_test(NAME fm COMMAND test_fm)
    add_test(NAME options COMMAND test_options)
    add_test(NAME spectrum COMMAND test_spectrum)
    add_test(NAME synth COMMAND test_synth)
    add_test(NAME gui COMMAND test_gui)
    add_test(NAME app COMMAND test_app)
    set_tests_properties(app PROPERTIES TIMEOUT 10)
    add_test(NAME help COMMAND keyboard_synth --help)
endif()
