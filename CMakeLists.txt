cmake_minimum_required(VERSION 3.16)
project(keyboard_synth VERSION 0.1.0 LANGUAGES C)

option(CSYNTH_BUILD_APP "Build the SDL synth window" ON)
option(CSYNTH_BUILD_SDL_ADAPTER "Build the optional SDL audio adapter" ${CSYNTH_BUILD_APP})
if(CSYNTH_BUILD_APP AND NOT CSYNTH_BUILD_SDL_ADAPTER)
    message(FATAL_ERROR "The SDL app needs CSYNTH_BUILD_SDL_ADAPTER=ON")
endif()
include(CTest)
add_subdirectory(libcsynth)

function(app_compile_options target)
    set_target_properties(${target} PROPERTIES C_EXTENSIONS OFF)
    target_compile_features(${target} PRIVATE c_std_11)
    target_include_directories(${target} PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4)
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
    endif()
endfunction()

if(CSYNTH_BUILD_SDL_ADAPTER)
    find_package(SDL2 2.0.18 REQUIRED CONFIG)
    add_library(csynth_sdl STATIC synth_sdl.c)
    add_library(csynth::sdl ALIAS csynth_sdl)
    target_include_directories(csynth_sdl PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
    target_link_libraries(csynth_sdl PUBLIC csynth::csynth SDL2::SDL2)
    app_compile_options(csynth_sdl)
endif()

if(CSYNTH_BUILD_APP)
    if(APPLE)
        set(CMAKE_FIND_FRAMEWORK LAST)
    endif()
    find_package(Freetype REQUIRED)
    add_executable(keyboard_synth main.c options.c spectrum.c midi.c file_browser.c spectrogram.c)
    target_link_libraries(keyboard_synth PRIVATE csynth::sdl Freetype::Freetype)
    if(TARGET SDL2::SDL2main)
        target_link_libraries(keyboard_synth PRIVATE SDL2::SDL2main)
    endif()
    target_compile_definitions(keyboard_synth PRIVATE SYNTH_DEFAULT_PRESET_DIR="${CMAKE_CURRENT_SOURCE_DIR}/libcsynth/presets")
    app_compile_options(keyboard_synth)
endif()

if(BUILD_TESTING)
    add_executable(test_midi tests/test_midi.c midi.c file_browser.c)
    add_executable(test_options tests/test_options.c options.c)
    add_executable(test_spectrum tests/test_spectrum.c spectrum.c)
    target_link_libraries(test_options PRIVATE csynth::csynth)
    foreach(test_name IN ITEMS midi options spectrum)
        app_compile_options(test_${test_name})
        if(UNIX)
            target_link_libraries(test_${test_name} PRIVATE m)
        endif()
        add_test(NAME ${test_name} COMMAND test_${test_name})
    endforeach()
    if(CSYNTH_BUILD_SDL_ADAPTER)
        add_executable(test_synth tests/test_synth.c spectrum.c)
        target_link_libraries(test_synth PRIVATE csynth::sdl)
        target_compile_definitions(test_synth PRIVATE SDL_MAIN_HANDLED)
        app_compile_options(test_synth)
        add_test(NAME synth COMMAND test_synth)
    endif()
    if(CSYNTH_BUILD_APP)
        add_executable(test_app tests/test_app.c options.c spectrum.c midi.c file_browser.c spectrogram.c)
        # test_gui includes spectrogram.c to inspect event dispatch state.
        add_executable(test_gui tests/test_gui.c midi.c file_browser.c spectrum.c)
        target_compile_definitions(test_gui PRIVATE SDL_MAIN_HANDLED)
        foreach(test_target IN ITEMS test_gui test_app)
            target_link_libraries(${test_target} PRIVATE csynth::sdl Freetype::Freetype)
            app_compile_options(${test_target})
        endforeach()
        add_test(NAME gui COMMAND test_gui)
        add_test(NAME app COMMAND test_app)
        set_tests_properties(app PROPERTIES TIMEOUT 10)
        add_test(NAME help COMMAND keyboard_synth --help)
    endif()
endif()
